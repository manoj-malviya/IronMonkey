```csharp
private async Task AddStage()
{
    Content.PipelineStages.Add(new PipelineStageDefinition
    {
        Name = "New Stage",
        Order = Content.PipelineStages.Count,
        StageType = "Active"
    });
    StateHasChanged();
    await NotifyChanged();
}

private async Task RemoveStage(int index)
{
    Content.PipelineStages.RemoveAt(index);
    // Re-assign order values
    for (int i = 0; i < Content.PipelineStages.Count; i++)
        Content.PipelineStages[i].Order = i;
    StateHasChanged();
    await NotifyChanged();
}
```

**OnNameChanged handler** — auto-generates IndustrySlug from Name unless admin has manually edited it:
```csharp
private void OnNameChanged(ChangeEventArgs e)
{
    var name = e.Value?.ToString() ?? string.Empty;
    _model.Name = name;
    if (!_slugManuallyEdited)
    {
        _model.IndustrySlug = GenerateSlug(name);
    }
}

private static string GenerateSlug(string name) =>
    System.Text.RegularExpressions.Regex.Replace(
        name.ToLowerInvariant().Trim(), @"[^a-z0-9]+", "-").Trim('-');
```

Also track manual slug edit via `@oninput="() => _slugManuallyEdited = true"` on the IndustrySlug InputText.

**HandleSaveAsync:**
```csharp
private async Task HandleSaveAsync()
{
    _isLoading = true;
    _errorBanner = null;

    try
    {
        var client = HttpClientFactory.CreateClient("AdminApi");
        var payload = new
        {
            _model.Name,
            _model.Description,
            _model.IconIdentifier,
            _model.IndustrySlug,
            _model.IsBlank,
            _model.Content
        };
        var response = await client.PostAsJsonAsync("/api/recipes", payload);

        if (response.IsSuccessStatusCode)
        {
            var result = await response.Content.ReadFromJsonAsync<RecipeCreateResponse>();
            Nav.NavigateTo($"/admin/recipes/{result?.RecipeId}/edit");
        }
        else if (response.StatusCode == System.Net.HttpStatusCode.Unauthorized)
        {
            Nav.NavigateTo("/login");
        }
        else
        {
            var errorBody = await response.Content.ReadAsStringAsync();
            _errorBanner = $"Failed to save recipe: {errorBody}";
        }
    }
    catch (Exception ex)
    {
        _errorBanner = $"Unable to save recipe: {ex.Message}";
    }
    finally
    {
        _isLoading = false;
    }
}

private record RecipeCreateResponse(Guid RecipeId, string Message);
```
