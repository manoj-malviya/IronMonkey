**@code block:**

```csharp
@code {
    private readonly SignupModel _model = new();
    private bool _isLoading = false;
    private string? _errorBanner = null;

    // Recipe browser state
    private List<RecipeListItem> _recipes = [];
    private Guid _selectedRecipeId = Guid.Empty;  // Guid.Empty = blank selected (default)
    private RecipePreviewPanel.RecipePreviewResponse? _selectedRecipePreview = null;
    private bool _isLoadingPreview = false;

    protected override async Task OnInitializedAsync()
    {
        try
        {
            var client = HttpClientFactory.CreateClient("AdminApi");
            var recipes = await client.GetFromJsonAsync<List<RecipeListItem>>("/api/recipes");
            // API returns blank recipe first (IsBlank=true ordered first), then by name
            // Filter out IsBlank from the main list (we render blank manually as first card)
            _recipes = recipes?.Where(r => !r.IsBlank).ToList() ?? [];
        }
        catch
        {
            // Silently fail — recipe browser is optional (SIGN-02 says "optionally select one")
        }
    }

    private async Task HandleRecipeClick(Guid id)
    {
        // Toggle off if already selected
        if (_selectedRecipeId == id)
        {
            _selectedRecipeId = Guid.Empty;
            _selectedRecipePreview = null;
            return;
        }

        _selectedRecipeId = id;
        _selectedRecipePreview = null;

        // Blank = Guid.Empty, no preview needed
        if (id == Guid.Empty) return;

        _isLoadingPreview = true;
        StateHasChanged();
        try
        {
            var client = HttpClientFactory.CreateClient("AdminApi");
            _selectedRecipePreview = await client.GetFromJsonAsync<RecipePreviewPanel.RecipePreviewResponse>($"/api/recipes/{id}");
        }
        catch
        {
            // Preview failed — recipe is still selected, just no content shown
        }
        finally
        {
            _isLoadingPreview = false;
        }
    }

    private string BlankCardClass => _selectedRecipeId == Guid.Empty
        ? "border-2 border-indigo-600"
        : "border border-slate-200";

    private async Task HandleSignupAsync()
    {
        _isLoading = true;
        _errorBanner = null;

        try
        {
            var client = HttpClientFactory.CreateClient("AdminApi");
            var response = await client.PostAsJsonAsync("/auth/signup", new
            {
                _model.CompanyName,
                _model.AdminEmail,
                _model.AdminPassword,
                _model.Phone,
                RecipeId = _selectedRecipeId == Guid.Empty ? (Guid?)null : _selectedRecipeId,
                _model.CompanySize,
                _model.Address,
                _model.BillingContact
            });

            if (response.StatusCode == System.Net.HttpStatusCode.Created)
            {
                Nav.NavigateTo("/signup/success");
            }
            else if (response.StatusCode == System.Net.HttpStatusCode.BadRequest)
            {
                _errorBanner = "Unable to create account. Please check your details and try again.";
            }
            else if (response.StatusCode == System.Net.HttpStatusCode.Conflict ||
                     (int)response.StatusCode == 422)
            {
                _errorBanner = "An account with this email already exists.";
            }
            else
            {
                _errorBanner = "Unable to create account. Please try again.";
            }
        }
        catch
        {
            _errorBanner = "Unable to reach the server. Please try again.";
        }
        finally
        {
            _isLoading = false;
        }
    }

    // Response record for GET /api/recipes list items
    private record RecipeListItem(
        Guid Id,
        string Name,
        string Description,
        string IconIdentifier,
        string IndustrySlug,
        bool IsBlank,
        int Version,
        int StageCount,
        int FieldCount,
        int RuleCount,
        int RoleCount);

    // Model for the EditForm with DataAnnotations validation
    private class SignupModel
    {
        [System.ComponentModel.DataAnnotations.Required(ErrorMessage = "Company name is required")]
        public string CompanyName { get; set; } = string.Empty;

        [System.ComponentModel.DataAnnotations.Required(ErrorMessage = "A valid admin email is required")]
        [System.ComponentModel.DataAnnotations.EmailAddress(ErrorMessage = "Enter a valid email address")]
        public string AdminEmail { get; set; } = string.Empty;

        [System.ComponentModel.DataAnnotations.Required(ErrorMessage = "Password must be at least 8 characters")]
        [System.ComponentModel.DataAnnotations.MinLength(8, ErrorMessage = "Password must be at least 8 characters")]
        public string AdminPassword { get; set; } = string.Empty;

        [System.ComponentModel.DataAnnotations.Required(ErrorMessage = "Phone is required")]
        public string Phone { get; set; } = string.Empty;

        [System.ComponentModel.DataAnnotations.Required(ErrorMessage = "Company size is required")]
        public string CompanySize { get; set; } = string.Empty;

        [System.ComponentModel.DataAnnotations.Required(ErrorMessage = "Address is required")]
        public string Address { get; set; } = string.Empty;

        [System.ComponentModel.DataAnnotations.Required(ErrorMessage = "Billing contact is required")]
        public string BillingContact { get; set; } = string.Empty;
    }
}
```

**Important notes:**
- The `ValidationError` from POST /auth/signup uses FluentValidation. The API returns 400 with problem details. Treat any non-201 as an error and show the banner.
- The duplicate email check in SignupRequestEndpoint returns a `ValidationError` (which the framework maps to 400 or 422 — confirm actual status code from the endpoint). Use the "An account with this email already exists." banner text per the UI-SPEC copywriting contract.
- `_selectedRecipeId = Guid.Empty` means "blank" is selected by default (per D-07). No RecipeId is sent when blank.
- `GetFromJsonAsync<RecipePreviewPanel.RecipePreviewResponse>` works because RecipePreviewPanel's nested record is public.
