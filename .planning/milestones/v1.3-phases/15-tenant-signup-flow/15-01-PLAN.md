Create `IronMonkey.Web/Components/Shared/RecipePreviewCard.razor` (create the Shared/ directory if it does not already exist).

The component renders a clickable card for recipe selection. Full file contents:

```razor
@* RecipePreviewCard — clickable recipe selection card for the signup flow *@
@* Per D-05, D-06: click to select, indigo-600 border when selected *@

<button type="button"
        @onclick="HandleClick"
        class="@CardClass w-full text-left rounded-lg bg-white px-4 py-6 shadow-sm cursor-pointer hover:opacity-95 transition-all">

    @if (!string.IsNullOrEmpty(IndustrySlug))
    {
        <span class="text-xs font-medium text-slate-500 uppercase tracking-wide">@IndustrySlug</span>
    }

    <h3 class="mt-1 text-base font-semibold text-slate-900">@Name</h3>

    @if (!string.IsNullOrEmpty(Description))
    {
        <p class="mt-1 text-sm text-slate-600 line-clamp-2">@Description</p>
    }

    <p class="mt-3 text-sm text-slate-500">
        @StageCount stages · @FieldCount fields · @RuleCount rules
    </p>

</button>

@code {
    [Parameter, EditorRequired] public Guid Id { get; set; }
    [Parameter, EditorRequired] public string Name { get; set; } = string.Empty;
    [Parameter] public string Description { get; set; } = string.Empty;
    [Parameter] public string IndustrySlug { get; set; } = string.Empty;
    [Parameter] public int StageCount { get; set; }
    [Parameter] public int FieldCount { get; set; }
    [Parameter] public int RuleCount { get; set; }
    [Parameter] public bool IsSelected { get; set; }
    [Parameter] public EventCallback<Guid> OnClick { get; set; }

    private string CardClass => IsSelected
        ? "border-2 border-indigo-600"
        : "border border-slate-200";

    private async Task HandleClick()
    {
        await OnClick.InvokeAsync(Id);
    }
}
```

Notes:
- `type="button"` prevents the button from triggering form submission when inside EditForm
- `line-clamp-2` requires Tailwind's line-clamp plugin; if not available use `overflow-hidden` with a fixed height as fallback — check IronMonkey.Web/package.json or tailwind config first and use whichever works
- Border styling is toggled via computed `CardClass` property (per D-06)

Create `IronMonkey.Web/Components/Shared/RecipePreviewPanel.razor`.

This panel displays the full recipe content below the selected recipe card. It receives a `RecipePreviewResponse` record (defined in Signup.razor's @code block and passed as a parameter). Since Blazor components cannot reference records defined in other pages' @code blocks, define a matching nested record here OR accept a parameter-object approach.

Use this approach: define the response record inline in this component so it's self-contained. Signup.razor will pass the deserialized content.

Full file contents:

```razor
@* RecipePreviewPanel — expanded recipe content below selected recipe card *@
@* Displays pipeline stages, custom fields, workflow rules, and roles *@

@if (Recipe is not null)
{
    <div class="mt-4 rounded-lg border border-slate-200 bg-slate-50 px-4 py-6">

        @* Pipeline Stages *@
        @if (Recipe.Content?.PipelineStages?.Count > 0)
        {
            <div class="mb-5">
                <h4 class="text-sm font-medium text-slate-700 mb-2">Pipeline Stages</h4>
                <ol class="space-y-1">
                    @foreach (var stage in Recipe.Content.PipelineStages.OrderBy(s => s.Order))
                    {
                        <li class="text-sm text-slate-600 flex items-center gap-2">
                            <span class="w-5 h-5 rounded-full bg-indigo-100 text-indigo-700 text-xs flex items-center justify-center font-medium flex-shrink-0">
                                @(stage.Order + 1)
                            </span>
                            @stage.Name
                            <span class="text-xs text-slate-400">(@stage.StageType)</span>
                        </li>
                    }
                </ol>
            </div>
        }

        @* Custom Fields *@
        @if (Recipe.Content?.CustomFields?.Count > 0)
        {
            <div class="mb-5">
                <h4 class="text-sm font-medium text-slate-700 mb-2">Custom Fields</h4>
                <div class="overflow-x-auto">
                    <table class="w-full text-sm">
                        <thead>
                            <tr class="text-xs text-slate-500 border-b border-slate-200">
                                <th class="text-left pb-1 font-medium">Field</th>
                                <th class="text-left pb-1 font-medium">Type</th>
                                <th class="text-left pb-1 font-medium">Required</th>
                            </tr>
                        </thead>
                        <tbody class="divide-y divide-slate-100">
                            @foreach (var field in Recipe.Content.CustomFields)
                            {
                                <tr>
                                    <td class="py-1 text-slate-700">@field.FieldName</td>
                                    <td class="py-1 text-slate-500">@field.FieldType</td>
                                    <td class="py-1 text-slate-500">@(field.IsRequired ? "Yes" : "No")</td>
                                </tr>
                            }
                        </tbody>
                    </table>
                </div>
            </div>
        }

        @* Workflow Rules *@
        @if (Recipe.Content?.WorkflowRules?.Count > 0)
        {
            <div class="mb-5">
                <h4 class="text-sm font-medium text-slate-700 mb-2">Workflow Rules</h4>
                <ul class="space-y-1">
                    @foreach (var rule in Recipe.Content.WorkflowRules)
                    {
                        <li class="text-sm text-slate-600">
                            @rule.Name
                            <span class="text-xs text-slate-400 ml-1">(on @rule.Trigger)</span>
                        </li>
                    }
                </ul>
            </div>
        }

        @* Roles *@
        @if (Recipe.Content?.Roles?.Count > 0)
        {
            <div>
                <h4 class="text-sm font-medium text-slate-700 mb-2">Roles</h4>
                <ul class="space-y-1">
                    @foreach (var role in Recipe.Content.Roles)
                    {
                        <li class="text-sm text-slate-600">
                            <span class="font-medium text-slate-700">@role.Name</span>
                            @if (!string.IsNullOrEmpty(role.Description))
                            {
                                <span class="text-slate-500 ml-1">— @role.Description</span>
                            }
                        </li>
                    }
                </ul>
            </div>
        }

    </div>
}

@code {
    [Parameter] public RecipePreviewResponse? Recipe { get; set; }

    // Mirrors RecipePreviewEndpoint.Response + RecipeContentModel

    public record RecipePreviewResponse(
        Guid Id,
        string Name,
        string Description,
        RecipeContent? Content);

    public record RecipeContent(
        List<PipelineStage>? PipelineStages,
        List<CustomField>? CustomFields,
        List<WorkflowRule>? WorkflowRules,
        List<Role>? Roles);

    public record PipelineStage(string Name, int Order, string StageType);
    public record CustomField(string FieldName, string FieldType, bool IsRequired, List<string>? Options);
    public record WorkflowRule(string Name, string Trigger, string ConditionJson, string ActionJson);
    public record Role(string Name, string Description);
}
```
