    // D-14: Two-step approve + provision chain
    private async Task HandleApproveAsync()
    {
        if (_actionSignupId is null) return;
        _isApproving = true;
        _errorBanner = null;
        _successMessage = null;
        try
        {
            var client = HttpClientFactory.CreateClient("AdminApi");
            var signupId = _actionSignupId.Value;

            // Step 1: Approve signup request
            var approveResponse = await client.PostAsJsonAsync(
                $"/admin/signup/{signupId}/approve",
                new { ApprovalNote = string.IsNullOrWhiteSpace(_approvalNote) ? (string?)null : _approvalNote });

            if (!approveResponse.IsSuccessStatusCode)
            {
                _errorBanner = $"Approval failed (HTTP {(int)approveResponse.StatusCode}).";
                return;
            }

            // Update row to Approved immediately
            var signupItem = _signups.FirstOrDefault(s => s.Id == signupId);
            if (signupItem is not null) signupItem.Status = "Approved";

            // Step 2: Provision tenant (D-14 chaining)
            var provisionResponse = await client.PostAsJsonAsync(
                $"/admin/tenants/{signupId}/provision", new { });

            if (provisionResponse.IsSuccessStatusCode)
            {
                // Both steps succeeded (D-16)
                if (signupItem is not null) signupItem.IsProvisioned = true;
                _successMessage = "Signup approved and tenant provisioned successfully.";
            }
            else
            {
                // D-17: Step 1 succeeded, Step 2 failed — show retry button
                _errorBanner = "Approval succeeded but provisioning failed. Use the 'Provision' button to retry.";
            }

            _showApproveModal = false;
            _approvalNote = string.Empty;
            _actionSignupId = null;
        }
        catch (Exception ex)
        {
            _errorBanner = $"Error: {ex.Message}";
        }
        finally
        {
            _isApproving = false;
        }
    }

    // D-15: Reject with required reason
    private async Task HandleRejectAsync()
    {
        if (_actionSignupId is null) return;

        // Front-end validation: reason is required (D-15)
        if (string.IsNullOrWhiteSpace(_rejectionReason))
        {
            _rejectValidationError = "Rejection reason is required.";
            return;
        }

        _isRejecting = true;
        _rejectValidationError = null;
        _errorBanner = null;
        _successMessage = null;
        try
        {
            var client = HttpClientFactory.CreateClient("AdminApi");
            var signupId = _actionSignupId.Value;

            var response = await client.PostAsJsonAsync(
                $"/admin/signup/{signupId}/reject",
                new { RejectionReason = _rejectionReason });

            if (response.IsSuccessStatusCode)
            {
                // D-16: Optimistic UI update
                var item = _signups.FirstOrDefault(s => s.Id == signupId);
                if (item is not null) item.Status = "Rejected";
                _successMessage = "Signup request rejected.";
                _showRejectModal = false;
                _rejectionReason = string.Empty;
                _actionSignupId = null;
            }
            else
            {
                _errorBanner = $"Rejection failed (HTTP {(int)response.StatusCode}).";
            }
        }
        catch (Exception ex)
        {
            _errorBanner = $"Error: {ex.Message}";
        }
        finally
        {
            _isRejecting = false;
        }
    }

    // D-17: Retry provision for Approved-but-not-provisioned entries
    private async Task RetryProvisionAsync(Guid id)

            _errorBanner = $"Error: {ex.Message}";
        }
