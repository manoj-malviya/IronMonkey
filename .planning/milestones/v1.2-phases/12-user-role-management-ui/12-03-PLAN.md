```razor
<div class="rounded-lg bg-green-50 border border-green-200 p-4">
    <div class="flex items-start justify-between mb-2">
        <p class="text-sm font-semibold text-green-800">User created successfully!</p>
        <button type="button" @onclick="OnClosed"
                class="text-slate-400 hover:text-slate-600 text-lg leading-none">&times;</button>
    </div>
    <p class="text-sm text-green-700 mb-2">Share this password (displayed once):</p>
    <div class="flex items-center gap-2">
        <code class="flex-1 rounded bg-white border border-green-200 px-3 py-2 text-sm font-mono text-slate-900">@Password</code>
        <button type="button" @onclick="CopyToClipboardAsync"
                class="rounded-lg border border-slate-300 px-3 py-2 text-sm font-medium text-slate-700 hover:bg-slate-50">
            @(_copied ? "Copied!" : "Copy")
        </button>
    </div>
</div>

@code {
    [Parameter] public string Password { get; set; } = string.Empty;
    [Parameter] public EventCallback OnClosed { get; set; }

    private bool _copied = false;

    private async Task CopyToClipboardAsync()
    {
        try
        {
            await JS.InvokeVoidAsync("navigator.clipboard.writeText", Password);
            _copied = true;
            // Reset after 2 seconds (fire and forget)
            _ = Task.Delay(2000).ContinueWith(_ => { _copied = false; InvokeAsync(StateHasChanged); });
        }
        catch
        {
            // clipboard API may not be available in all browsers — silent fail
        }
    }
}
```

The `OnClosed` callback is invoked by the parent to dismiss/hide this component — the component itself just shows the X button which fires the callback. The parent controls whether the component is rendered (using a `_generatedPassword != null` flag).

For the **Reset Password** use case on UserEdit, the heading can be different — but since UserPasswordDisplay is shared and the difference is minor, use the same "User created successfully!" text (the context makes it clear). The alternative of a "Password reset successfully!" variant is left to Claude's discretion — either add a second `[Parameter] public string Heading { get; set; } = "User created successfully!"` or keep it fixed. Recommended: add `[Parameter] public string Heading { get; set; } = "User created successfully!"` for reusability, use "Password reset successfully!" when rendered from reset password action.

**UserCreate @code:**
```csharp
private UserFormModel _model = new();
private int _selectedRoleId = 0;
private List<RoleItem> _roles = [];
private string? _generatedPassword = null;
private bool _isLoading = false;
private bool _submitted = false;
private string? _errorBanner = null;

protected override async Task OnInitializedAsync() => await LoadRolesAsync();

private async Task LoadRolesAsync()
{
    try
    {
        var client = HttpClientFactory.CreateClient("AdminApi");
        var roles = await client.GetFromJsonAsync<List<RoleItem>>("/api/user-management/roles");
        _roles = roles ?? [];
    }
    catch (Exception ex)
    {
        _errorBanner = $"Failed to load roles: {ex.Message}";
    }
}

private async Task HandleSaveAsync()
{
    _submitted = true;
    if (_selectedRoleId == 0) return; // role not selected

    _isLoading = true;
    _errorBanner = null;
    try
    {
        var client = HttpClientFactory.CreateClient("AdminApi");
        var payload = new { _model.Name, _model.Email, RoleId = _selectedRoleId };
        var response = await client.PostAsJsonAsync("/api/user-management/users", payload);
        if (response.IsSuccessStatusCode)
        {
            var result = await response.Content.ReadFromJsonAsync<CreateUserResponse>();
            _generatedPassword = result?.GeneratedPassword;
        }
        else if (response.StatusCode == System.Net.HttpStatusCode.Unauthorized)
            Nav.NavigateTo("/login");
        else
        {
            var body = await response.Content.ReadAsStringAsync();
            _errorBanner = $"Failed to save user: {body}";
        }
    }
    catch (Exception ex) { _errorBanner = $"Unable to save user: {ex.Message}"; }
    finally { _isLoading = false; }
}

private void HandlePasswordDisplayClosed()
{
    Nav.NavigateTo("/admin/users");
}

private record CreateUserResponse(Guid UserId, string GeneratedPassword, string Message);
private record RoleItem(int RoleId, string RoleName);

private class UserFormModel
{
    [Required(ErrorMessage = "Name is required.")]
    public string Name { get; set; } = string.Empty;

    [Required(ErrorMessage = "Email is required.")]
    [EmailAddress(ErrorMessage = "Email must be a valid email address.")]
    public string Email { get; set; } = string.Empty;
}
```

**UserEdit @code:**
```csharp
[Parameter] public Guid Id { get; set; }

private UserFormModel _model = new();
private int _selectedRoleId = 0;
private List<RoleItem> _roles = [];
private string _userName = string.Empty;
private string? _generatedPassword = null;
private bool _isLoading = true;
private bool _isSaving = false;
private bool _isResettingPassword = false;
private bool _submitted = false;
private string? _errorBanner = null;

protected override async Task OnInitializedAsync()
{
    await Task.WhenAll(LoadUserAsync(), LoadRolesAsync());
}

private async Task LoadUserAsync()
{
    try
    {
        var client = HttpClientFactory.CreateClient("AdminApi");
        var user = await client.GetFromJsonAsync<UserDetail>($"/api/user-management/users/{Id}");
        if (user is not null)
        {
            _model.Name = user.Name;
            _model.Email = user.Email;
            _selectedRoleId = user.RoleId;
            _userName = user.Name;
        }
    }
    catch (Exception ex) { _errorBanner = $"Failed to load user: {ex.Message}"; }
    finally { _isLoading = false; }
}

private async Task LoadRolesAsync()
{
    try
    {
        var client = HttpClientFactory.CreateClient("AdminApi");
        var roles = await client.GetFromJsonAsync<List<RoleItem>>("/api/user-management/roles");
        _roles = roles ?? [];
    }
    catch (Exception ex) { _errorBanner = $"Failed to load roles: {ex.Message}"; }
}

private async Task HandleSaveAsync()
{
    _submitted = true;
    if (_selectedRoleId == 0) return;

    _isSaving = true;
    _errorBanner = null;
    try
    {
        var client = HttpClientFactory.CreateClient("AdminApi");
        var payload = new { _model.Name, _model.Email, RoleId = _selectedRoleId };
        var response = await client.PutAsJsonAsync($"/api/user-management/users/{Id}", payload);
        if (response.IsSuccessStatusCode)
        {
            _userName = _model.Name; // update title
        }
        else if (response.StatusCode == System.Net.HttpStatusCode.Unauthorized)
            Nav.NavigateTo("/login");
        else
        {
            var body = await response.Content.ReadAsStringAsync();
            _errorBanner = $"Failed to save user: {body}";
        }
    }
    catch (Exception ex) { _errorBanner = $"Unable to save user: {ex.Message}"; }
    finally { _isSaving = false; }
}

private async Task HandleResetPasswordAsync()
{
    _isResettingPassword = true;
    _errorBanner = null;
    try
    {
        var client = HttpClientFactory.CreateClient("AdminApi");
        var response = await client.PostAsync($"/api/user-management/users/{Id}/reset-password", null);
        if (response.IsSuccessStatusCode)
        {
            var result = await response.Content.ReadFromJsonAsync<ResetPasswordResponse>();
            _generatedPassword = result?.GeneratedPassword;
        }
        else if (response.StatusCode == System.Net.HttpStatusCode.Unauthorized)
            Nav.NavigateTo("/login");
        else
        {
            var body = await response.Content.ReadAsStringAsync();
            _errorBanner = $"Failed to reset password: {body}";
        }
    }
    catch (Exception ex) { _errorBanner = $"Unable to reset password: {ex.Message}"; }
    finally { _isResettingPassword = false; }
}

private record UserDetail(Guid Id, string Name, string Email, int RoleId, string RoleName, bool IsActive, DateTime CreatedAt);
private record RoleItem(int RoleId, string RoleName);
private record ResetPasswordResponse(string GeneratedPassword, string Message);

private class UserFormModel
{
    [Required(ErrorMessage = "Name is required.")]
    public string Name { get; set; } = string.Empty;

    [Required(ErrorMessage = "Email is required.")]
    [EmailAddress(ErrorMessage = "Email must be a valid email address.")]
    public string Email { get; set; } = string.Empty;
}
```
