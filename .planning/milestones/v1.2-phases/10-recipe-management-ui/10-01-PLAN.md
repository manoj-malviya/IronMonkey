private List<RecipeListItem> _recipes = [];
private bool _isLoading = true;
private bool _showInactive = false;
private string? _errorBanner = null;
private string? _successMessage = null;
private bool _isDeactivating = false;

private List<RecipeListItem> DisplayedRecipes =>

**OnInitializedAsync:** Calls `await LoadRecipesAsync()`.

**LoadRecipesAsync:**
```csharp
private async Task LoadRecipesAsync()
{
    _isLoading = true;
    _errorBanner = null;
    try
    {
        var client = HttpClientFactory.CreateClient("AdminApi");
        var list = await client.GetFromJsonAsync<List<RecipeListItem>>("/api/recipes");
        _recipes = list ?? [];
    }
    catch (Exception ex)
    {
        _errorBanner = $"Failed to load recipes: {ex.Message}";
    }
    finally
    {
        _isLoading = false;
    }
}
```

**HandleDeactivateAsync:**
```csharp
private async Task HandleDeactivateAsync(Guid id)
{
    _isDeactivating = true;
    _deactivatingId = id;
    _errorBanner = null;
    _successMessage = null;
    try
    {
        var client = HttpClientFactory.CreateClient("AdminApi");
        var response = await client.DeleteAsync($"/api/recipes/{id}");
        if (response.IsSuccessStatusCode)
        {
            var item = _recipes.FirstOrDefault(r => r.Id == id);
            if (item is not null) item.IsActive = false;
            _successMessage = "Recipe deactivated successfully.";
        }
        else if (response.StatusCode == System.Net.HttpStatusCode.Unauthorized)
        {
            Nav.NavigateTo("/login");
        }
        else
        {
            _errorBanner = "Failed to deactivate recipe.";
        }
    }
    catch (Exception ex)
    {
        _errorBanner = $"Error: {ex.Message}";
    }
    finally
    {
        _isDeactivating = false;
        _deactivatingId = null;
    }
}
```
